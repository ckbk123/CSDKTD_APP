cmake_minimum_required(VERSION 3.24)
project(CSDKTD_APP VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(CSDKTD_BUILD_TESTS "Build catalogue and navigation tests" ON)
option(CSDKTD_GUI_TESTS "Register the Vulkan shell smoke test with CTest" OFF)
set(CSDKTD_RAVEN_SOURCE_DIR "" CACHE PATH "Use a local Raven source checkout")

set(RAVEN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(RAVEN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RAVEN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(RAVEN_BUILD_WEB OFF CACHE BOOL "" FORCE)
set(RAVEN_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(RAVEN_MSVC_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
if(CSDKTD_RAVEN_SOURCE_DIR)
    set(CSD_RAVEN_DIR "${CSDKTD_RAVEN_SOURCE_DIR}")
    add_subdirectory("${CSDKTD_RAVEN_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/raven")
else()
    include(FetchContent)
    FetchContent_Declare(raven
        GIT_REPOSITORY https://github.com/oscar-soirey/Raven.git
        GIT_TAG c4410662723bf0eec633a36a7c2e4d69490325cf)
    FetchContent_MakeAvailable(raven)
    set(CSD_RAVEN_DIR "${raven_SOURCE_DIR}")
endif()
# Build/link the needed Raven target without installing its development SDK.
set_property(DIRECTORY "${CSD_RAVEN_DIR}" PROPERTY EXCLUDE_FROM_ALL TRUE)

add_library(csd_module_api INTERFACE)
target_include_directories(csd_module_api INTERFACE "${PROJECT_SOURCE_DIR}/include")
target_link_libraries(csd_module_api INTERFACE raven::raven)
add_subdirectory(modules)

# Embed the existing university logo so moving the executable cannot lose it.
file(READ "${PROJECT_SOURCE_DIR}/assets/hcmut.png" CSD_LOGO_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1," CSD_LOGO_BYTES "${CSD_LOGO_HEX}")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${CMAKE_BINARY_DIR}/generated/csd_logo.h"
    "static const unsigned char csd_logo_data[] = {${CSD_LOGO_BYTES}};\n")

add_library(csd_shell STATIC src/shell.c src/module_registry.c)
target_include_directories(csd_shell PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(csd_shell PUBLIC csd_module_api csd_modules)
if(MSVC)
    target_compile_options(csd_shell PRIVATE /W4 /utf-8)
else()
    target_compile_options(csd_shell PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(csdktd_app src/main.c)
target_link_libraries(csdktd_app PRIVATE csd_shell)
set_target_properties(csdktd_app PROPERTIES OUTPUT_NAME "CSDKTD APP")
if(WIN32)
    enable_language(RC)
    target_sources(csdktd_app PRIVATE assets/app.rc)
    target_link_libraries(csdktd_app PRIVATE user32)
    # Keep a console in Debug; Release is a normal desktop application.
    set_target_properties(csdktd_app PROPERTIES WIN32_EXECUTABLE "$<NOT:$<CONFIG:Debug>>")
elseif(APPLE)
    set_target_properties(csdktd_app PROPERTIES MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_GUI_IDENTIFIER "vn.edu.hcmut.csdktd"
        MACOSX_BUNDLE_BUNDLE_NAME "CSDKTD APP")
endif()

if(CSDKTD_BUILD_TESTS)
    enable_testing()
    add_executable(csdktd_catalog_test tests/catalog_test.c src/module_registry.c)
    target_link_libraries(csdktd_catalog_test PRIVATE csd_module_api csd_modules)
    add_test(NAME module_catalog COMMAND csdktd_catalog_test)
    add_test(NAME module_navigation COMMAND csdktd_app --smoke-test)
    # A stateful replacement for CR01 checks the module lifecycle contract.
    add_executable(csdktd_lifecycle_test tests/lifecycle_test.c)
    target_link_libraries(csdktd_lifecycle_test PRIVATE csd_shell)
    add_test(NAME module_lifecycle COMMAND csdktd_lifecycle_test)
    if(CSDKTD_GUI_TESTS)
        add_test(NAME single_window_module_navigation COMMAND csdktd_app --gui-smoke-test)
    endif()
endif()

include(GNUInstallDirs)
install(TARGETS csdktd_app RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}" BUNDLE DESTINATION .)
install(FILES "${PROJECT_SOURCE_DIR}/THIRD_PARTY_NOTICES.md" DESTINATION .)
install(FILES "${PROJECT_SOURCE_DIR}/docs/RUNNING.txt" DESTINATION . RENAME README.txt)
install(FILES "${PROJECT_SOURCE_DIR}/assets/README.md" DESTINATION licenses RENAME HCMUT-ASSET.md)
install(FILES "${CSD_RAVEN_DIR}/LICENSE" DESTINATION licenses RENAME Raven-LICENSE)
install(FILES "${CSD_RAVEN_DIR}/NOTICE" "${CSD_RAVEN_DIR}/THIRD_PARTY.md" DESTINATION licenses)
install(FILES "${CSD_RAVEN_DIR}/third_party/stb/LICENSE" DESTINATION licenses RENAME stb-LICENSE)
install(FILES "${CSD_RAVEN_DIR}/third_party/roboto/LICENSE" DESTINATION licenses RENAME Roboto-LICENSE)
install(FILES "${CSD_RAVEN_DIR}/third_party/roboto-mono/LICENSE" DESTINATION licenses RENAME Roboto-Mono-LICENSE)
